create table Customer_address(
addr_id varchar2(5) check (addr_id like 'A%') primary key,
city varchar2(20) not null,
state varchar2(20) not null,
pincode number(6) not null
);

create table customer(
c_id varchar2(5) check (c_id like 'C%') primary key,
dob date not null,
fname varchar2(20) not null,
mname varchar2(20),
lname varchar2(20) not null,
addr_id varchar2(5) references Customer_address(addr_id) on delete cascade not null
);

create table Customer_ph_no(
c_id varchar2(5) references Customer(c_id) on delete cascade not null,
ph_no number(10),
primary key (c_id, ph_no)
);

create table car(
car_lno varchar(20) primary key,
c_id varchar2(5) references Customer(c_id) on delete cascade not null,
company varchar2(20) not null,
model varchar2(20) not null
);

create table FirstHand(
car_lno varchar2(20) references car(car_lno) on delete cascade not null,
ownership_type varchar2(1) default 'F' check (ownership_type='F'),
primary key (car_lno)
);

create table SecondHand(
car_lno varchar2(20) references car(car_lno) on delete cascade not null,
ownership_type varchar2(1) default 'F' check (ownership_type='F' or ownership_type='S'),
primary key (car_lno)
);

create table Accident(
fir_no varchar2(10) check (fir_no like 'FIR%') primary key,
car_lno varchar2(20) references car(car_lno) on delete cascade not null,
accident_date date not null,
accident_loc varchar2(20) not null,
damage_amt number(8, 2) not null
);

1. Find the total number of people who owned cars that were involved in accidents in 2010.
select count(*) as number_of_people from
customer join car on customer.c_id = car.c_id
join accident on car.car_lno = accident.car_lno
where extract(year from accident.accident_date) = 2010;

2. Find the number of accidents in which the cars belonging to “XYZ” were involved.
select count(*) as number_of_accidents from
customer join car on customer.c_id = car.c_id
join accident on car.car_lno = accident.car_lno
where customer.fname = 'XYZ';
